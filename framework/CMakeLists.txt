option(ECHO_FRAMEWORK_ENABLE_DATA "Build JSON, content and save services" ON)
option(ECHO_FRAMEWORK_BUILD_TESTS "Build headless Echo Framework tests" OFF)
option(ECHO_FRAMEWORK_BUILD_EXAMPLE "Build the headless core demonstration" OFF)
option(ECHO_FRAMEWORK_BUILD_TOOLS "Build native host-side framework tools" OFF)
option(ECHO_FRAMEWORK_ENABLE_NARRATIVE "Build the bounded narrative/dialogue module" ON)
option(ECHO_FRAMEWORK_ENABLE_POWER "Build the bounded power circuit module" ON)

add_subdirectory(core)

if(ECHO_FRAMEWORK_ENABLE_DATA)
    include(FetchContent)
    set(YYJSON_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(YYJSON_BUILD_FUZZER OFF CACHE BOOL "" FORCE)
    set(YYJSON_BUILD_MISC OFF CACHE BOOL "" FORCE)
    set(YYJSON_BUILD_DOC OFF CACHE BOOL "" FORCE)
    set(YYJSON_INSTALL OFF CACHE BOOL "" FORCE)
    set(YYJSON_DISABLE_FILE ON CACHE BOOL "" FORCE)
    set(YYJSON_DISABLE_INCR_READER ON CACHE BOOL "" FORCE)
    set(YYJSON_DISABLE_UTILS ON CACHE BOOL "" FORCE)
    set(YYJSON_READER_DEPTH_LIMIT 64 CACHE STRING "" FORCE)
    set(YYJSON_WRITER_DEPTH_LIMIT 64 CACHE STRING "" FORCE)
    FetchContent_Declare(
        yyjson
        GIT_REPOSITORY https://github.com/ibireme/yyjson.git
        GIT_TAG 6447536015f3d600f3d65323b10976103b337ca7
        GIT_SHALLOW FALSE
    )
    FetchContent_MakeAvailable(yyjson)
    add_subdirectory(data)
    add_subdirectory(abilities)
    add_subdirectory(inventory)
    add_subdirectory(attributes)
    add_subdirectory(equipment)
    add_subdirectory(quests)
    add_subdirectory(cutscenes)
    add_subdirectory(world)
    if(ECHO_FRAMEWORK_ENABLE_POWER)
        add_subdirectory(power)
    endif()
endif()

if(ECHO_FRAMEWORK_ENABLE_POWER AND NOT ECHO_FRAMEWORK_ENABLE_DATA)
    message(FATAL_ERROR "The power module requires ECHO_FRAMEWORK_ENABLE_DATA.")
endif()

if(ECHO_FRAMEWORK_ENABLE_NARRATIVE)
    if(NOT ECHO_FRAMEWORK_ENABLE_DATA)
        message(FATAL_ERROR "The narrative module requires ECHO_FRAMEWORK_ENABLE_DATA.")
    endif()
    add_subdirectory(narrative)
endif()

if(ECHO_FRAMEWORK_BUILD_TESTS)
    add_subdirectory(tests)
endif()

if(ECHO_FRAMEWORK_BUILD_EXAMPLE)
    if(NOT ECHO_FRAMEWORK_ENABLE_DATA)
        message(FATAL_ERROR "The headless example requires ECHO_FRAMEWORK_ENABLE_DATA.")
    endif()
    add_subdirectory(examples/core_demo)
    if(TARGET EchoFramework::Narrative)
        add_subdirectory(examples/narrative_demo)
    endif()
endif()

if(ECHO_FRAMEWORK_BUILD_TOOLS)
    if(NOT ECHO_FRAMEWORK_ENABLE_DATA)
        message(FATAL_ERROR "Framework tools require ECHO_FRAMEWORK_ENABLE_DATA.")
    endif()
    if(CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "Framework tools must be built with a native host toolchain.")
    endif()
    add_subdirectory(tools)
endif()
