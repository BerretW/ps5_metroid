cmake_minimum_required(VERSION 3.24)
project(echo_ps5_platform LANGUAGES C)
if(NOT PS5 OR NOT CMAKE_HOST_WIN32)
    message(FATAL_ERROR "Use the experimental native Windows PS5 toolchain.")
endif()
file(TO_CMAKE_PATH "${PLATFORM_SOURCE}" PLATFORM_SOURCE)
if(NOT EXISTS "${PLATFORM_SOURCE}/platform/Makefile")
    message(FATAL_ERROR "Pinned PS5 platform source is required.")
endif()
file(GLOB sources CONFIGURE_DEPENDS "${PLATFORM_SOURCE}/platform/src/*.c")
file(GLOB regex_sources CONFIGURE_DEPENDS "${PLATFORM_SOURCE}/platform/src/regex/*.c")
set(heap_source "${PLATFORM_SOURCE}/platform/src/dlmalloc/mspace.c")
add_library(ps5platform STATIC ${sources} ${regex_sources} "${heap_source}")
set_target_properties(ps5platform PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_include_directories(ps5platform PRIVATE "${PLATFORM_SOURCE}/platform/include")
target_compile_options(ps5platform PRIVATE -O2 -fPIC -Wall -Wextra -Werror)
set_source_files_properties(${regex_sources} "${heap_source}" PROPERTIES COMPILE_OPTIONS "-w")
